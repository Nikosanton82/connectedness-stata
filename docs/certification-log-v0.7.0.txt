. program define cn_dcccase
  1.     syntax varlist, case(string) [opts(string)]
  2.     local f "fixtures/R_`case'_par.csv"
  3.     capture confirm file "`f'"
  4.     if _rc {
  5.         di as txt "  R_`case': no fixture, skipped"
  6.         exit
  7.     }
  8.     local k : word count `varlist'
  9.     tempname P E

 13.         quietly import delimited using "`f'", clear varnames(1) asdouble
 14.         mkmat _all, matrix(`P')
 15.     }
 16.     frame drop __fx
 17.     local llh = `P'[1, 4*`k'+3]
 18.     matrix __rpar = `P'[1, 1..4*`k'+2]

> (cn_test) replace notable

 21.     global cn_tests = $cn_tests + 1

 23.     local ok = (`d' < 1e-10)
 24.     if !`ok' global cn_fails = $cn_fails + 1
 25.     di as txt "  " %-26s "R_`case'" " log likelihood " as res %9.2e `d' as 
> txt "  " cond(`ok', "pass", "FAIL")
 26.     connectedness `varlist', model(dccgarch) `opts' frame(cn_test) replace 
> notable

 32.     if !`ok' global cn_fails = $cn_fails + 1
 33.     di as txt "  " %-26s "R_`case'" " estimates " as res %9.2e `d' as txt "
